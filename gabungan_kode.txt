export default function PasswordField({
  id,
  label,
  value,
  onChange,
  placeholder = "",
}) {
  const [showPassword, setShowPassword] = useState(false);

  return (
    <Form.Group className="mb-3">
      <Form.Label htmlFor={id}>
        {label} <span className="required">*</span>
      </Form.Label>

      <div className="password-wrapper">
        <Form.Control
          id={id}
          name={id}
          type={showPassword ? "text" : "password"}
          value={value}
          onChange={onChange}
          placeholder={placeholder}
          required
        />

        <button
          type="button"
          className="password-toggle"
          onClick={() => setShowPassword(!showPassword)}
          aria-label={showPassword ? "Sembunyikan password" : "Tampilkan password"}
        >
          <i className={`bi ${showPassword ? "bi-eye-slash" : "bi-eye"}`} />
        </button>
      </div>
    </Form.Group>
  );

export default function AuthForm({ type }) {
  const isLogin = type === "login";

  const [formData, setFormData] = useState({
    name: "",
    email: "",
    gender: "",
    phone: "",
    password: "",
    confirmPassword: "",
  });

  const [dialCode, setDialCode] = useState("+62");

  const handleChange = (event) => {
    const { name, value } = event.target;

    setFormData((previous) => ({
      ...previous,
      [name]: value,
    }));
  };

  const handleSubmit = (event) => {
    event.preventDefault();

    if (
      !isLogin &&
      formData.password !== formData.confirmPassword
    ) {
      alert("Konfirmasi kata sandi tidak sama.");
      return;
    }

    alert(
      isLogin
        ? "Login berhasil untuk prototype statis."
        : "Registrasi berhasil untuk prototype statis."
    );
  };

  return (
    <div className={isLogin ? "auth-card login-card" : "auth-card register-card"}>
      <h1>{isLogin ? "Masuk ke Akun" : "Pendaftaran Akun"}</h1>

      <p className="subtitle">
        {isLogin
          ? "Yuk, lanjutin belajarmu di videobelajar"
          : "Yuk, daftarkan akunmu sekarang juga!"}
      </p>

      <Form onSubmit={handleSubmit}>
        {!isLogin && (
          <Form.Group className="mb-3">
            <Form.Label htmlFor="name">
              Nama Lengkap <span className="required">*</span>
            </Form.Label>

            <Form.Control
              id="name"
              name="name"
              type="text"
              value={formData.name}
              onChange={handleChange}
              required
            />
          </Form.Group>
        )}

        <Form.Group className="mb-3">
          <Form.Label htmlFor={`${type}-email`}>
            Email <span className="required">*</span>
          </Form.Label>

          <Form.Control
            id={`${type}-email`}
            name="email"
            type="email"
            value={formData.email}
            onChange={handleChange}
            placeholder="Masukkan email"
            autoComplete="email"
            required
          />
        </Form.Group>

        {!isLogin && (
          <>
            <Form.Group className="mb-3 gender-field">
              <Form.Label htmlFor="gender">
                Jenis Kelamin <span className="required">*</span>
              </Form.Label>

              <Form.Select
                id="gender"
                name="gender"
                value={formData.gender}
                onChange={handleChange}
                required
              >
                <option value="">Pilih Jenis Kelamin</option>
                <option value="pria">Pria</option>
                <option value="wanita">Wanita</option>
              </Form.Select>
            </Form.Group>

            <Form.Group className="mb-3">
              <Form.Label htmlFor="phone">
                No. Hp <span className="required">*</span>
              </Form.Label>

              <div className="phone-group">
                <Form.Select
                  value={dialCode}
                  onChange={(event) => setDialCode(event.target.value)}
                  className="dial-code"
                >
                  <option value="+62">🇮🇩 +62</option>
                  <option value="+65">🇸🇬 +65</option>
                  <option value="+60">🇲🇾 +60</option>
                  <option value="+66">🇹🇭 +66</option>
                </Form.Select>

                <Form.Control
                  id="phone"
                  name="phone"
                  type="tel"
                  value={formData.phone}
                  onChange={handleChange}
                  placeholder="Masukkan nomor HP"
                  required
                />
              </div>
            </Form.Group>
          </>
        )}

        <PasswordField
          id={`${type}-password`}
          label={isLogin ? "Password" : "Kata Sandi"}
          value={formData.password}
          onChange={(event) =>
            setFormData((previous) => ({
              ...previous,
              password: event.target.value,
            }))
          }
          placeholder="Masukkan password"
        />

        {!isLogin && (
          <PasswordField
            id="confirm-password"
            label="Konfirmasi Kata Sandi"
            value={formData.confirmPassword}
            onChange={(event) =>
              setFormData((previous) => ({
                ...previous,
                confirmPassword: event.target.value,
              }))
            }
            placeholder="Ulangi password"
          />
        )}

        <div className="form-action">
          <a href="#forgot-password">Lupa Password?</a>
        </div>

        <Button type="submit" className="primary-auth-button">
          {isLogin ? "Masuk" : "Daftar"}
        </Button>

        <Button
          as={Link}
          to={isLogin ? "/register" : "/login"}
          type="button"
          className="secondary-auth-button"
        >
          {isLogin ? "Daftar" : "Masuk"}
        </Button>

        <div className="divider">
          <span>Atau</span>
        </div>

        <Button type="button" className="google-button">
          <i className="bi bi-google" />
          Masuk dengan Google
        </Button>
      </Form>
    </div>
  );
